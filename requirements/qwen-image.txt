# venvs/qwenimage — Qwen-Image-2.1 unified still-image adapter
# (job type `qwen-image`, model id `qwen-image-2.1`)
#
# Recreate ON SPARK (venvs/ is spark-only state, never deployed by
# deploy-to-spark.sh; the venv MUST be a real directory, never a symlink —
# worker policy rejects symlinked venv paths):
#
#   cd ~/src/arbiter
#   python3.12 -m venv --copies venvs/qwenimage
#   venvs/qwenimage/bin/pip install --upgrade pip wheel
#   venvs/qwenimage/bin/pip install torch --index-url https://download.pytorch.org/whl/cu130
#   venvs/qwenimage/bin/pip install "transformers==5.17.0" accelerate pillow
#   venvs/qwenimage/bin/pip install "diffusers @ git+https://github.com/huggingface/diffusers@80c7ed262aeffbeb43ef13ae04baeb9b84515a69"
#   cp venvs/qwentts/lib/python3.12/site-packages/arbiter.pth venvs/qwenimage/lib/python3.12/site-packages/
#
# Notes:
# - torch 2.14.0+cu130 aarch64 from the PyTorch cu130 index (GB10 / CUDA 13).
#   (The index serves the latest cu130 build; 2.12/2.14 are both known-good.)
# - transformers 5.17.0 is the first release carrying the Qwen3-VL pieces the
#   QwenImage21Pipeline text encoder needs; 5.8.x (flux2 venv) is too old.
# - diffusers is pinned to a git commit because QwenImage21Pipeline is NOT in
#   any released diffusers yet (0.40.0 predates it). Re-pin to a release once
#   one ships QwenImage21Pipeline.
# - Weights (~33 GB bf16: 7B DiT + Qwen3-VL text encoder + VAE) live in the
#   HF cache on /mnt/t9 (symlinked ~/.cache/huggingface/hub). auto_download:
#   "Qwen/Qwen-Image-2.1" in spark local/config.json records the checkpoint.
torch==2.14.0+cu130
transformers==5.17.0
accelerate==1.13.0
pillow
diffusers @ git+https://github.com/huggingface/diffusers@80c7ed262aeffbeb43ef13ae04baeb9b84515a69
